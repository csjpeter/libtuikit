# Declares the test suites of libtuikit and their helper programs.

# Test suites.  One executable per suite; the kind is the CTest label, so
# "ctest -L unit" and "ctest -L pty" run one kind each.
# See policies/testing-model.md.
#
# libptytest is a dependency of the tests only: the library itself shares
# no code and no logic with it (docs/specification.md, section 2).



set(TUIKIT_TEST_TIMEOUT 180)

# A command every test is run under, e.g. Valgrind (see manage.sh valgrind).
set(TUIKIT_TEST_WRAPPER "" CACHE STRING "Command prefix for every test")

# Fixtures written by the suites go under the build directory, never into
# a fixed /tmp path (policies/project-boundaries-model.md).
set(TUIKIT_TEST_TMPDIR "${CMAKE_CURRENT_BINARY_DIR}/tmp")
file(MAKE_DIRECTORY "${TUIKIT_TEST_TMPDIR}")

find_library(TUIKIT_UTIL_LIB util)

# Every suite is linked with the fault-injection wrappers (common/faults.c):
# inert until a test arms them, and the raise() wrapper writes the coverage
# counters before a signal ends a process.  Test binaries only.
set(TUIKIT_WRAPPED malloc calloc realloc pipe fcntl tcgetattr tcsetattr
                   poll read __read_chk write raise)
set(TUIKIT_WRAP_FLAGS "")
foreach(_fn IN LISTS TUIKIT_WRAPPED)
    list(APPEND TUIKIT_WRAP_FLAGS "LINKER:--wrap=${_fn}")
endforeach()

# Where the tests are built: <root>/<category>/bin/<name>, one file per
# declared test, whether compiled or a script.  manage.sh points the test
# tree (build/_test) at build/tests, the place the integrator runs them from
# (policies/testing-model.md, "How a test is declared, built and found");
# the other trees keep theirs inside themselves.
set(TUIKIT_TEST_BIN_ROOT "${CMAKE_CURRENT_BINARY_DIR}" CACHE PATH
    "Directory that receives <category>/bin/<name> for every declared test")

# project_test(<category> <name> SOURCES <file>... [LIB <library>] [TIMEOUT <s>])
# project_test(<category> <name> SCRIPT <file> [ARGS <arg>...] [TIMEOUT <s>])
#
# Every test is declared on one line of this shape, read by the integrator
# without building anything; unit/check_tests.sh fails on a declared test
# with no file in bin/ and on a file in bin/ that is not declared.
#
# A compiled test is linked with the fault-injection wrappers
# (common/faults.c): inert until a test arms them, and the raise() wrapper
# writes the coverage counters before a signal ends a process.  A script
# gets a small wrapper in bin/ that runs it with its arguments, so that
# everything in bin/ runs by its path alone.
function(project_test _cat _name)
    cmake_parse_arguments(T "" "SCRIPT;LIB;TIMEOUT" "SOURCES;ARGS" ${ARGN})
    set(_bin "${TUIKIT_TEST_BIN_ROOT}/${_cat}/bin")
    if(NOT T_TIMEOUT)
        set(T_TIMEOUT ${TUIKIT_TEST_TIMEOUT})
    endif()
    if(T_SOURCES)
        if(NOT T_LIB)
            set(T_LIB tuikit_static)
        endif()
        add_executable(${_name} ${T_SOURCES} "${CMAKE_CURRENT_SOURCE_DIR}/common/faults.c")
        set_target_properties(${_name} PROPERTIES RUNTIME_OUTPUT_DIRECTORY "${_bin}")
        target_link_options(${_name} PRIVATE ${TUIKIT_WRAP_FLAGS})
        if(ENABLE_COVERAGE)
            target_compile_definitions(${_name} PRIVATE TUIKIT_COVERAGE_BUILD)
        endif()
        target_include_directories(${_name} PRIVATE "${CMAKE_CURRENT_SOURCE_DIR}/common")
        target_compile_options(${_name} PRIVATE -Wall -Wextra -Wshadow)
        target_link_libraries(${_name} PRIVATE ${T_LIB} ${TUIKIT_PTYTEST})
        if(TUIKIT_UTIL_LIB)
            target_link_libraries(${_name} PRIVATE util)   # openpty() on older glibc
        endif()
        target_compile_definitions(${_name} PRIVATE
            TUIKIT_TEST_TMPDIR="${TUIKIT_TEST_TMPDIR}")
        add_test(NAME ${_cat}.${_name} COMMAND ${TUIKIT_TEST_WRAPPER} "${_bin}/${_name}")
    elseif(T_SCRIPT)
        set(_cmd "'${CMAKE_CURRENT_SOURCE_DIR}/${T_SCRIPT}'")
        foreach(_a IN LISTS T_ARGS)
            string(APPEND _cmd " '${_a}'")
        endforeach()
        file(GENERATE OUTPUT "${_bin}/${_name}"
             CONTENT "#!/bin/sh\n# Generated by tests/CMakeLists.txt: runs ${T_SCRIPT}.\nexec ${_cmd} \"$@\"\n"
             FILE_PERMISSIONS OWNER_READ OWNER_WRITE OWNER_EXECUTE
                              GROUP_READ GROUP_EXECUTE WORLD_READ WORLD_EXECUTE)
        add_test(NAME ${_cat}.${_name} COMMAND "${_bin}/${_name}")
    else()
        message(FATAL_ERROR "project_test(${_cat} ${_name}): SOURCES or SCRIPT is required")
    endif()
    set_tests_properties(${_cat}.${_name} PROPERTIES LABELS ${_cat} TIMEOUT ${T_TIMEOUT})
endfunction()
